    def to_text(self) -> str:
    """A plain terminal report: summary + spans + steps + LLM calls."""
    lines = [f"trace {self.id}"]
    for key, value in self._summary():
        if key != "run":
            lines.append(f"  {key:<10} {value}")
    for span in self.spans:
        lines.append(
            f"  span {span.agent} on {span.event_artifact_type or '-'}"
            f" ({span.latency_ms:.1f} ms)"
        )
        for step in span.produces:
            lines.append(
                f"    step {step.name}  ops={step.operations}"
                f"  {step.latency_ms:.1f} ms"
            )
        for call in span.llm_calls:
            lines.append(
                f"    llm  {call.model or call.provider or '-'}"
                f"  {call.prompt_tokens}+{call.completion_tokens} tok"
                f"  {call.latency_ms:.1f} ms"
            )
        for write in span.writes:
            lines.append(f"    write {write.data_type} {write.artifact_id}")
    return "\n".join(lines)

    def to_markdown(self) -> str:
    """Markdown report for a PR comment or CI summary."""
    lines = [f"## Trace `{self.id}`", ""]
    lines.append("| field | value |")
    lines.append("| --- | --- |")
    for key, value in self._summary():
        lines.append(f"| {key} | `{value}` |")
    if not self.spans:
        return "\n".join(lines) + "\n\n_No spans._"
    lines.extend(["", "### Spans", ""])
    for span in self.spans:
        lines.append(
            f"- **{span.agent}** on `{span.event_artifact_type or '-'}`"
            f" — {span.latency_ms:.1f} ms"
        )
        for step in span.produces:
            lines.append(
                f"  - step `{step.name}` ({step.operations} ops,"
                f" {step.latency_ms:.1f} ms)"
            )
        for call in span.llm_calls:
            lines.append(
                f"  - LLM `{call.model or call.provider or '-'}`"
                f" ({call.prompt_tokens}+{call.completion_tokens} tok,"
                f" {call.latency_ms:.1f} ms)"
            )
        for write in span.writes:
            lines.append(f"  - write `{write.data_type}` `{write.artifact_id}`")
    return "\n".join(lines) + "\n"

    def to_html(self) -> str:
    """A self-contained HTML report (no server, no external assets)."""
    rows = "".join(
        f"<tr><th>{html.escape(key)}</th><td>{html.escape(value)}</td></tr>"
        for key, value in self._summary()
    )
    spans: list[str] = []
    for span in self.spans:
        steps = "".join(
            f"<li><code>{html.escape(step.name)}</code>"
            f" — {step.operations} ops, {step.latency_ms:.1f} ms</li>"
            for step in span.produces
        )
        calls = "".join(
            f"<li><code>{html.escape(call.model or call.provider or '-')}</code>"
            f" — {call.prompt_tokens}+{call.completion_tokens} tok,"
            f" {call.latency_ms:.1f} ms</li>"
            for call in span.llm_calls
        )
        writes = "".join(
            f"<li><code>{html.escape(write.data_type)}</code>"
            f" <code>{html.escape(write.artifact_id)}</code></li>"
            for write in span.writes
        )
        body = ""
        if steps:
            body += f"<h4>Steps</h4><ul>{steps}</ul>"
        if calls:
            body += f"<h4>LLM calls</h4><ul>{calls}</ul>"
        if writes:
            body += f"<h4>Writes</h4><ul>{writes}</ul>"
        spans.append(
            f"<details open><summary><b>{html.escape(span.agent)}</b>"
            f" on <code>{html.escape(span.event_artifact_type or '-')}</code>"
            f" — {span.latency_ms:.1f} ms</summary>{body}</details>"
        )
    return (
        '<!doctype html><html><head><meta charset="utf-8">'
        f"<title>Trace {html.escape(self.id)}</title>"
        "<style>body{font:14px system-ui,sans-serif;max-width:900px;"
        "margin:2rem auto;padding:0 1rem;color:#111}"
        "table{border-collapse:collapse}th,td{text-align:left;padding:.25rem .75rem;"
        "border-bottom:1px solid #ddd}code{background:#f3f3f3;padding:.1rem .25rem}"
        "details{margin:.5rem 0;padding:.5rem;border:1px solid #ddd;border-radius:6px}"
        "summary{cursor:pointer}h4{margin:.5rem 0 .25rem}</style></head><body>"
        f"<h1>Trace <code>{html.escape(self.id)}</code></h1>"
        f"<table>{rows}</table>{''.join(spans)}</body></html>"
    )

    StepRun¶

    StepRun ¶

    Bases: BaseModel

    One step execution within an agent span (§54).

    An agent runs one or more steps per event; the span records which ones actually ran and how much each contributed (operations = authored effect operations, so the step that did the work is distinguishable from one that merely matched the event).

    Tag¶

    Tag ¶

    Bases: BaseModel

    A vocabulary entry: a named label humans use to mark runs for review.

    Tags are annotations, not runtime data: the runtime never produces them, a reviewer adds them from the dashboard after inspecting a trace (e.g. bad-prompt, hallucination, needs-review). count is filled in by list_tags (how many runs carry the tag) and left at 0 elsewhere.

    TagAssignment¶

    TagAssignment ¶

    Bases: BaseModel

    One tag attached to one run, with an optional reviewer note.

    The note is free text describing what is wrong / what to change — the payload a reviewer leaves so the run can be found again by tag and acted on (prompt/behaviour fix). color is copied from the vocabulary for rendering.

    TraceLimits¶

    TraceLimits ¶

    Bases: BaseModel

    Write-time limits for one recorded run.

    Configure on the tracer (Tracer(store=..., limits=TraceLimits(...))), not on RuntimeResources: these are recording policy, independent of the app's providers/sources. When a cap is hit, extra data is dropped and the trace is marked trace_truncated instead of growing without bound.

    ToolRun¶

    ToolRun ¶

    Bases: BaseModel

    One tool execution recorded for a trace span (AgentSpan.tools).

    ToolUse/run_tool_loop open a per-task run log via begin_tool_runs(); the runtime attaches it to the agent span. Args are deliberately not recorded (they may carry secrets); name/latency/status/size is enough to spot a slow or failing tool.

    async def children(
    self, run_id: str, *, limit: int = 50, offset: int = 0
) -> list[dict[str, Any]]:
    """Child runs of `run_id` (subflow/branches), oldest first."""
    result = await self.query(

        limit=limit,
        offset=offset,
    )
    return list(result["items"])

    async def stream(self, request: LLMRequest) -> AsyncIterator[LLMResponseChunk]:
    """Yields the provider's chunks and records one `LLMCall` at the end.

    The extra layer is deliberately cheap: chunks are appended to a list
    (never `text +=`), the buffer is capped at `truncate`, and `_record`
    runs once — in `finally` — not per chunk. When `on_delta` is set, each
    chunk is forwarded (sync, best-effort) for a live UI.
    """
    started_at = datetime.now(UTC)
    started = time.monotonic()
    parts: list[str] = []
    size = 0
    limit = self._truncate
    truncated = False
    usage: dict[str, Any] = {}
    error: str | None = None
    try:
        async for chunk in self._inner.stream(request):
            if chunk.usage:
                usage.update(chunk.usage)
            if chunk.text:
                if limit is None:
                    parts.append(chunk.text)
                elif size < limit:
                    take = chunk.text[: limit - size]
                    parts.append(take)
                    size += len(take)
                    if len(take) < len(chunk.text):
                        truncated = True
                else:
                    truncated = True
            if chunk.text:
                self._safe_delta(chunk.text, streamed=True)
            yield chunk
    except asyncio.CancelledError:
        error = "cancelled"
        raise
    except Exception as exc:  # noqa: BLE001 — record and re-raise
        error = f"{type(exc).__name__}: {exc}"
        raise
    finally:
        text = "".join(parts)
        if truncated and limit is not None:
            # `_record` clips to `limit` and adds the ellipsis itself;
            # passing the marker through keeps that single clip honest.
            text += "…"
        self._safe_record(
            request,
            LLMResponse(text=text, usage=usage),
            (time.monotonic() - started) * 1000,
            error,
            started_at,
        )

    RunTracer ¶

    RunTracer(
    context: Context,
    tracer: Tracer | CompositeTracer | None,

    Runtime's tracing collaborator (§54).

    Builds ArtifactRef/RelationRef/AgentSpan/RunTrace and delivers them to the configured Tracer, so Runtime only calls into this at a few well-defined points instead of interleaving span-building with the dispatch loop. Owns: wrapping resources.llm in RecordingLLM at turn start (on the runtime's per-turn resources copy), the task→agent attribution that wrapper needs to attribute an LLM call to the agent that made it, the per-turn span buffer, and the artifact-data truncation cache (ArtifactRef.data is memoized per (artifact_id, version) for one turn, since the same artifact is often referenced — as a read — by several spans in a generation).

    A no-op when tracer is None (the common case, no tracing configured): every method still works, record_span just returns None and nothing is buffered or sent anywhere.

    def record_span(
    self,
    agent: StepGroup,
    event: Event,
    reads: list[Read],
    latency_ms: float,
    *,
    error: BaseException | None = None,
    produces: list[tuple[str, int, float, datetime]] | None = None,
    tools: list[ToolRun] | None = None,
) -> AgentSpan | None:
    """Builds, buffers and delivers one span — or a no-op if disabled.

    `error is not None` covers the isolated-error path; the success path
    (a span whose `writes`/`relations` are filled in once the patch is
    applied, see `Runtime._commit_patches_to_apply`) passes it as `None`.
    `produces` is the ``(name, operations, latency_ms, started_at)`` list the agent's
    `execute()` recorded for this event.
    """
    if self.tracer is None:
        return None
    if self._max_spans is not None and len(self.spans) >= self._max_spans:
        self.truncated = True
        return None
    try:
        if self._max_refs is not None and len(reads) > self._max_refs:
            self.truncated = True
            reads = reads[: self._max_refs]
        llm_calls = self._pending_llm.pop(agent.name, [])
        if self._max_llm is not None and len(llm_calls) > self._max_llm:
            self.truncated = True
            llm_calls = llm_calls[: self._max_llm]
        if self._max_refs is not None and tools is not None:
            tools = tools[: self._max_refs]
        consumes = [
            getattr(c.artifact_type, "__name__", str(c.artifact_type))
            for c in (agent.consumes or [])
            if c.artifact_type is not None
        ]
        produces_declared = [
            getattr(p, "name", type(p).__name__) for p in (agent.steps or [])
        ]
        declared_tools = getattr(agent, "tools", ()) or ()
        if isinstance(declared_tools, dict):
            declared_tools = declared_tools.values()
        tools_declared = [
            getattr(t, "name", type(t).__name__) for t in declared_tools
        ]
        span = AgentSpan(
            agent=agent.name,
            event_type=event.type.value,
            event_artifact_type=(
                event.artifact_type.__name__
                if isinstance(event.artifact_type, type)
                else str(event.artifact_type)
            ),
            event_artifact_id=event.artifact_id,
            contract=AgentContract(
                consumes=consumes,
                produces=produces_declared,
                tools=tools_declared,
                priority=agent.priority,
            ),
            produces=[
                StepRun(
                    name=name,
                    operations=operations,
                    latency_ms=latency,
                    started_at=started_at,
                )
                for name, operations, latency, started_at in (produces or [])
            ],
            tools=tools or [],
            reads=self.read_refs(reads),
            latency_ms=latency_ms,
            llm_calls=llm_calls,
            error=(
                _redact(f"{type(error).__name__}: {error}", self._redactor)
                if error is not None
                else None
            ),
            # The span is recorded after its agent finished (delivery is
            # once per generation), so the true start is the record time
            # minus the measured latency — this keeps parallel agents'
            # bars side by side on the dashboard timeline.
            started_at=datetime.now(UTC) - timedelta(milliseconds=latency_ms),
        )
        self.spans.append(span)
        self.tracer.on_span(span)
        return span
    except Exception as exc:  # noqa: BLE001 — best-effort observability
        _swallow_tracing_error("on_span", exc)
        return None

    Observer of runs for the Runtime (§54).

    Runtime invokes on_turn_begin / on_agent_start / on_tool_start / on_tool_end / on_span / on_commit (sync, no I/O) and on_turn_end (async — the sinks write to SQLite/Postgres/Langfuse). on_turn_end awaits each sink's export, so sinks are async stores.

    def keep_trace(trace: RunTrace, sample: Sample) -> bool:
